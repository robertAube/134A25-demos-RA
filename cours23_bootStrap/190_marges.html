Différentes marges

border rounded

Aucune marge

m-5 border rounded

m-5 
m = marge sur les 4 côtés
5 = taille de la marge (échelle Bootstrap : 0 à 5)

ms-5 border rounded

ms-5 
m = margin (marge)
s = start (côté gauche)
4 = taille de la marge (échelle Bootstrap : 0 à 5)

mt-4 border rounded

mt-4 
m = margin (marge)
t = top (haut)
4 = taille de la marge (échelle Bootstrap : 0 à 5)

mb-4 border rounded

mb-4 
m = margin (marge)
t = bottom (bas)
4 = taille de la marge (échelle Bootstrap : 0 à 5)

mx-5 border rounded

mx-5 
m = margin (marge)
x = marge gauche et droite
4 = taille de la marge (échelle Bootstrap : 0 à 5)

my-5 border rounded

my-5 
m = margin (marge)
y = marge haut et bas
4 = taille de la marge (échelle Bootstrap : 0 à 5)
